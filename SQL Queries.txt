create database hr_analytics;
use hr_analytics;

create table hr_attrition (
Age INT,
Attrition VARCHAR(10),
BusinessTravel VARCHAR(50),
DailyRate INT,
Department VARCHAR(50),
DistanceFromHome INT,
Education INT,
EducationField VARCHAR(50),
EmployeeCount INT,
EmployeeNumber INT PRIMARY KEY,
EnvironmentSatisfaction INT,
Gender VARCHAR(10),
HourlyRate INT,
JobInvolvement INT,
JobLevel INT,
JobRole VARCHAR(50),
JobSatisfaction INT,
MaritalStatus VARCHAR(20),
MonthlyIncome INT,
MonthlyRate INT,
NumCompaniesWorked INT,
Over18 VARCHAR(5),
OverTime VARCHAR(5),
PercentSalaryHike INT,
PerformanceRating INT,
RelationshipSatisfaction INT,
StandardHours INT,
StockOptionLevel INT,
TotalWorkingYears INT,
TrainingTimesLastYear INT,
WorkLifeBalance INT,
YearsAtCompany INT,
YearsInCurrentRole INT,
YearsSinceLastPromotion INT,
YearsWithCurrManager INT
);

select * from hr_attrition;

select count(*) from hr_attrition;

select attrition, count(*)
from hr_attrition
group by attrition;


# core hr kpi's

-- total employees
select count(*) as TotalEmployees
from hr_attrition;

-- attrition count
select count(*) as EmployeesLeft
from hr_attrition
where attrition = "Yes";

-- attrition rate %
select sum(attrition = "yes") / count(*) * 100
from hr_attrition; 


-- department analysis

select department, count(*) as EmployeesLeft
from hr_attrition
where attrition = "yes"
group by department
order by EmployeesLeft desc;

-- salary impact
select 
case
when monthlyincome < 3000 then "Low Salary"
when monthlyincome between 3000 and 7000 then "Medium Salary"
else "High Salary"
end as Salary_Slab,
count(*) as attrition_count
from hr_attrition
where attrition = "yes"
group by salary_slab;


-- Experience Impact

select yearsAtCompany,
count(*) as EmployeesLeft
from hr_attrition
where attrition = "yes"
group by yearsAtCompany
order by yearsAtCompany;


-- Satisfaction Impact

select JobSatisfaction,
count(*) as EmployeesLeft
from hr_attrition
where attrition = "yes"
group by JobSatisfaction
order by JobSatisfaction desc;


###################################################################
-- Statistical analysis

-- Mean salary

select avg(MonthlyIncome) as avg_salary 
from hr_attrition;


-- attrition by overtime

select OverTime, count(*) as LeftEmployeesCount
from hr_attrition
where attrition = "yes"
group by overTime;
                -- or   
select OverTime, sum(attrition = "yes") as left_count
from hr_attrition
group by overTime;
